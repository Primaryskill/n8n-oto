Schema SQL PostgreSQL siap pakai
-- AutoPosting Med · Schema PostgreSQL
CREATE TABLE IF NOT EXISTS social_accounts (
  id            SERIAL PRIMARY KEY,
  platform      VARCHAR(20) NOT NULL CHECK (platform IN ('instagram','facebook')),
  username      VARCHAR(100) NOT NULL,
  ig_user_id    VARCHAR(50),
  fb_page_id    VARCHAR(50),
  access_token  TEXT NOT NULL,
  token_expires DATE,
  max_post_per_day INT DEFAULT 3,
  is_active     BOOLEAN DEFAULT true,
  created_at    TIMESTAMP DEFAULT NOW(),
  updated_at    TIMESTAMP DEFAULT NOW()
);
CREATE TABLE IF NOT EXISTS content_library (
  id         SERIAL PRIMARY KEY,
  title      VARCHAR(200),
  media_url  TEXT NOT NULL,
  caption    TEXT,
  hashtags   TEXT,
  is_active  BOOLEAN DEFAULT true,
  created_at TIMESTAMP DEFAULT NOW()
);
CREATE TABLE IF NOT EXISTS posting_schedule (
  id                SERIAL PRIMARY KEY,
  account_id        INTEGER REFERENCES social_accounts(id),
  content_id        INTEGER REFERENCES content_library(id),
  platform          VARCHAR(20) NOT NULL,
  post_date         DATE NOT NULL,
  post_session      VARCHAR(10) NOT NULL CHECK (post_session IN ('morning','evening')),
  delay_type        VARCHAR(10) DEFAULT 'random',
  account_order     INTEGER DEFAULT 1,
  status            VARCHAR(20) DEFAULT 'pending',
  campaign_end_date DATE,
  social_post_id    VARCHAR(100),
  error_message     TEXT,
  executed_at       TIMESTAMP,
  created_at        TIMESTAMP DEFAULT NOW(),
  updated_at        TIMESTAMP DEFAULT NOW(),
  UNIQUE(account_id, post_date, post_session)
);
CREATE TABLE IF NOT EXISTS posting_log (
  id            SERIAL PRIMARY KEY,
  schedule_id   INTEGER REFERENCES posting_schedule(id),
  account_id    INTEGER REFERENCES social_accounts(id),
  platform      VARCHAR(20),
  action        VARCHAR(50) DEFAULT 'auto_post',
  status        VARCHAR(20),
  post_id       VARCHAR(100),
  error_message TEXT,
  created_at    TIMESTAMP DEFAULT NOW()
);
CREATE INDEX idx_schedule_date   ON posting_schedule(post_date);
CREATE INDEX idx_schedule_status ON posting_schedule(status);
CREATE INDEX idx_log_created     ON posting_log(created_at DESC);
